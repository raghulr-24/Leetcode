class Solution:
    def search(self, nums: List[int], target: int) -> int:
        n = len(nums)
        l = 0
        r = n - 1
        while l < r:
            mid = (l + r) // 2
            if nums[mid] > nums[r]:
                l = mid + 1
            else:
                r = mid
        min_index = r
        if min_index == 0:
            l = 0
            r = n - 1
        elif target >= nums[0] and target <= nums[min_index -1]:
            l = 0 
            r = min_index - 1
        else:
            l = min_index 
            r = n - 1
        while l <= r:
            mid = (l + r) // 2
            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                l = mid + 1
            else:
                r = mid - 1
        return -1
