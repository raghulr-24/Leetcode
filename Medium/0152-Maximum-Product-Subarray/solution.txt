class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        res = max(nums)
        cmin = 1
        cmax = 1
        for n in nums:
            temp = cmax*n
            cmax = max(temp, cmin*n, n)
            cmin = min(temp, cmin*n, n)
            res = max(res, cmax)
        return res
